ERC-20 Transfer check + confirmations → unlock.

const { verifyPayment } = require('@josealbertomaquina/usdc-paywall')
const r = await verifyPayment({ txHash, wallet: MY_ADDR, amount: 3, chain: 'base' })
if (r.ok) res.send(premium)
else  res.status(402).json(r)  // Payment Required
